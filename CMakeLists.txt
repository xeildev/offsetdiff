cmake_minimum_required(VERSION 3.24)
project(xeil_offsetdiff VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
endif()

add_library(offsetdiff_core
  src/core/Parser.cpp src/core/Comparator.cpp src/core/Exporter.cpp)
target_include_directories(offsetdiff_core PUBLIC src PRIVATE third_party)
option(OFFSETDIFF_BUILD_GUI "Build the native Windows desktop application" ON)
if(WIN32 AND OFFSETDIFF_BUILD_GUI)
  add_executable(offsetdiff WIN32 src/main.cpp src/ui/Window.cpp src/ui/Renderer.cpp src/app.rc)
  target_link_libraries(offsetdiff PRIVATE offsetdiff_core d2d1 dwrite dwmapi comdlg32 comctl32 shell32 ole32)
  target_include_directories(offsetdiff PRIVATE third_party)
  target_compile_definitions(offsetdiff PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_link_options(offsetdiff PRIVATE /MANIFEST:NO)
  endif()
  set_target_properties(offsetdiff PROPERTIES OUTPUT_NAME "offsetdiff")
  install(TARGETS offsetdiff RUNTIME DESTINATION .)
  install(DIRECTORY examples DESTINATION .)
  install(FILES README.md LICENSE third_party/LICENSE.nlohmann DESTINATION .)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(offsetdiff_tests tests/CoreTests.cpp)
  target_link_libraries(offsetdiff_tests PRIVATE offsetdiff_core)
  target_include_directories(offsetdiff_tests PRIVATE third_party)
  add_test(NAME offsetdiff_core COMMAND offsetdiff_tests)
endif()
